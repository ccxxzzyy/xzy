3. 三元运算符 条件

var xzy_age = 20; var xzy_status = (xzy_age >= 18) ? "成年" : "未成年"; var xzy_num = 7; var xzy_parity = (xzy_num % 2 === 0) ? "偶数" : "奇数";

4. for 循环 循环

// 计算 1~10 的和 var xzy_sum = 0; for (var xzy_i = 1; xzy_i <= 10; xzy_i++) { xzy_sum += xzy_i; } // for 循环打印乘法口诀 var xzy_koujue = ""; for (var xzy_a = 1; xzy_a <= 9; xzy_a++) { for (var xzy_b = 1; xzy_b <= xzy_a; xzy_b++) { xzy_koujue += xzy_b + "×" + xzy_a + "=" + (xzy_a * xzy_b) + " "; } xzy_koujue += "\n"; }

5. while 循环 循环

// 数7游戏：1~30，跳过含7的数和7的倍数 var xzy_n = 1; var xzy_count = 0; var xzy_result = ""; while (xzy_n <= 30) { if (xzy_n % 7 === 0 || xzy_n % 10 === 7 || String(xzy_n).indexOf("7") !== -1) { xzy_result += "[" + xzy_n + "] "; xzy_count++; } else { xzy_result += xzy_n + " "; } xzy_n++; }

6. do...while 循环 循环

// 至少执行一次 var xzy_i = 0; var xzy_do_result = ""; do { xzy_do_result += xzy_i + " "; xzy_i += 2; } while (xzy_i < 10);

7. for...of 遍历数组 循环

var xzy_arr = ["苹果", "香蕉", "橘子", "葡萄"]; var xzy_fruits = ""; for (var xzy_f of xzy_arr) { xzy_fruits += xzy_f + " "; } // break 遇到 "橘子" 就跳出 var xzy_break_demo = ""; for (var xzy_f of xzy_arr) { if (xzy_f === "橘子") break; xzy_break_demo += xzy_f + " "; } // continue 跳过 "香蕉" var xzy_continue_demo = ""; for (var xzy_f of xzy_arr) { if (xzy_f === "香蕉") continue; xzy_continue_demo += xzy_f + " "; }
